// Add this match block inside your existing:
// service cloud.firestore { match /databases/{database}/documents { ... } }
//
// It assumes site administrators are identified by users/{uid}.role == "admin",
// matching the CourtRush application logic in index.html.

match /supportRequests/{requestId} {
  function isSiteAdmin() {
    return request.auth != null
      && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin";
  }

  allow create: if request.auth != null
    && request.resource.data.reporterUid == request.auth.uid
    && request.resource.data.status == "open"
    && request.resource.data.keys().hasOnly([
      "id",
      "reporterUid",
      "reporterPlayerId",
      "reporterName",
      "reporterEmail",
      "category",
      "subject",
      "message",
      "status",
      "createdAt",
      "updatedAt"
    ]);

  allow read: if request.auth != null
    && (resource.data.reporterUid == request.auth.uid || isSiteAdmin());

  allow update: if isSiteAdmin()
    && request.resource.data.reporterUid == resource.data.reporterUid
    && request.resource.data.createdAt == resource.data.createdAt;

  allow delete: if false;
}

// Add this helper near your other validation helpers.
function validPlayerDivision(value) {
  return value == "novice"
    || value == "beginner"
    || value == "intermediate"
    || value == "advanced"
    || value == "pro";
}

// Add "division" to the owner-editable fields in match /players/{playerId}.
// This lets each signed-in Club Member set only their own division.
allow update: if request.auth != null
  && resource.data.ownerUid == request.auth.uid
  && request.resource.data.ownerUid == request.auth.uid
  && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
    "name",
    "avatarUrl",
    "profileVisibility",
    "division",
    "clubId",
    "clubIds",
    "updatedAt"
  ])
  && validPlayerDivision(request.resource.data.get("division", "novice"));
